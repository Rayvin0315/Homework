// Speech recognition from microphone.
public static async Task RecognitionWithMicrophoneAsync()
{
    var config = SpeechConfig.FromSubscription("", "");

    using (var recognizer = new SpeechRecognizer(config))
    {
        while (true)
        {
            Console.WriteLine("Say something...");
            var result = await recognizer.RecognizeOnceAsync().ConfigureAwait(false);

            if (result.Reason == ResultReason.RecognizedSpeech)
            {
                var recognizedText = result.Text.ToLower();
                Console.WriteLine($"RECOGNIZED: Text={recognizedText}");

                if (recognizedText == "who are you?")
                {
                    await SpeechSynthesisSamples.SpeakTextAsync("It's a secret.");
                }
                else if (recognizedText == "who am i?")
                {
                    await SpeechSynthesisSamples.SpeakTextAsync("Rayvin.");
                }
                else if (recognizedText == "what's my favorite color?")
                {
                    await SpeechSynthesisSamples.SpeakTextAsync("Blue.");
                }
                else
                {
                    Console.WriteLine("Unrecognized command.");
                }
            }
            else if (result.Reason == ResultReason.NoMatch)
            {
                Console.WriteLine("NOMATCH: Speech could not be recognized.");
            }
            else if (result.Reason == ResultReason.Canceled)
            {
                var cancellation = CancellationDetails.FromResult(result);
                Console.WriteLine($"CANCELED: Reason={cancellation.Reason}");

                if (cancellation.Reason == CancellationReason.Error)
                {
                    Console.WriteLine($"CANCELED: ErrorCode={cancellation.ErrorCode}");
                    Console.WriteLine($"CANCELED: ErrorDetails={cancellation.ErrorDetails}");
                    Console.WriteLine($"CANCELED: Did you update the subscription info?");
                }
            }
        }
    }
}

        public static async Task SynthesisToSpeakerAsync()
        {
            var config = SpeechConfig.FromSubscription("", "");

            using (var synthesizer = new SpeechSynthesizer(config))
            {
                while (true)
                {
                    Console.WriteLine("Enter some text that you want to speak, or enter empty text to exit.");
                    Console.Write("> ");
                    string text = Console.ReadLine();
                    if (string.IsNullOrEmpty(text))
                    {
                        break;
                    }

                    using (var result = await synthesizer.SpeakTextAsync(text))
                    {
                        if (result.Reason == ResultReason.SynthesizingAudioCompleted)
                        {
                            Console.WriteLine($"Speech synthesized to speaker for text [{text}]");
                        }
                        else if (result.Reason == ResultReason.Canceled)
                        {
                            var cancellation = SpeechSynthesisCancellationDetails.FromResult(result);
                            Console.WriteLine($"CANCELED: Reason={cancellation.Reason}");

                            if (cancellation.Reason == CancellationReason.Error)
                            {
                                Console.WriteLine($"CANCELED: ErrorCode={cancellation.ErrorCode}");
                                Console.WriteLine($"CANCELED: ErrorDetails=[{cancellation.ErrorDetails}]");
                                Console.WriteLine($"CANCELED: Did you update the subscription info?");
                            }
                        }
                    }
                }
            }
        }

        // Method to speak text using synthesis.
        public static async Task SpeakTextAsync(string text)
        {
            var config = SpeechConfig.FromSubscription("", "");
            using (var synthesizer = new SpeechSynthesizer(config))
            {
                using (var result = await synthesizer.SpeakTextAsync(text))
                {
                    if (result.Reason == ResultReason.SynthesizingAudioCompleted)
                    {
                        // Optionally, you can log or process the successful synthesis here
                        Console.WriteLine($"Successfully synthesized text: {text}");
                    }
                    else if (result.Reason == ResultReason.Canceled)
                    {
                        var cancellation = SpeechSynthesisCancellationDetails.FromResult(result);
                        Console.WriteLine($"CANCELED: Reason={cancellation.Reason}");

                        if (cancellation.Reason == CancellationReason.Error)
                        {
                            Console.WriteLine($"CANCELED: ErrorCode={cancellation.ErrorCode}");
                            Console.WriteLine($"CANCELED: ErrorDetails=[{cancellation.ErrorDetails}]");
                            Console.WriteLine($"CANCELED: Did you update the subscription info?");
                        }
                    }
                }
            }
        }
